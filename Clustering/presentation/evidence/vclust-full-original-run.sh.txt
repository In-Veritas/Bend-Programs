#!/bin/sh
V=$HOME/Bend-Programs/Clustering/.tools/vclust-1.3.1/bin/vclust
F=$HOME/Bend-Programs/Clustering/datasets/infant_gut_virome/raw/all.curated.vOTUs.EarlyVir.fna
cd "$(dirname "$0")"
for t in 12 1; do
  /usr/bin/time -f "prefilter t=$t %e s %M KB" $V prefilter -i $F -o fltr$t.txt -t $t -v 0
  /usr/bin/time -f "align t=$t %e s %M KB" $V align -i $F -o ani$t.tsv --filter fltr$t.txt -t $t -v 0
done
for a in single complete uclust cd-hit set-cover leiden; do
  /usr/bin/time -f "cluster $a %e s %M KB" $V cluster -i ani12.tsv -o cl_$a.tsv --ids ani12.ids.tsv --algorithm $a --metric ani --ani 0.95 -v 0
done
echo DONE
